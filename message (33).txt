async function loadVersions() {
  try {
    const res = await fetch("https://piston-meta.mojang.com/mc/game/version_manifest_v2.json");
    const data = await res.json();

    VERSIONS.releases = [];
    VERSIONS.snapshots = [];
    VERSIONS.betas = [];
    VERSIONS.alphas = [];
    VERSIONS.old = [];

    data.versions.forEach(v => {
      const date = new Date(v.releaseTime).toLocaleDateString('ru-RU');

      if (v.type === "release") {
        VERSIONS.releases.push([v.id, date]);
      }
      else if (v.type === "snapshot") {
        VERSIONS.snapshots.push([v.id, date]);
      }
      else if (v.type === "old_beta") {
        VERSIONS.betas.push([v.id, date]);
      }
      else if (v.type === "old_alpha") {
        if (v.id.startsWith("c") || v.id.startsWith("rd-") || v.id.startsWith("inf-") || v.id.startsWith("0.31")) {
          VERSIONS.old.push([v.id, date]);
        }
        else if (v.id.startsWith("a")) {
          VERSIONS.alphas.push([v.id, date]);
        }
      }
    });

    VERSIONS.old.unshift(

function createInstance() {
  const name = document.getElementById('instanceName').value.trim();
  const group = document.getElementById('instanceGroup').value;
  const version = document.getElementById('instanceVersion').value;
  const loader = document.getElementById('instanceLoader').value;
  const ram = parseInt(document.getElementById('instanceRAM').value);
  
  if (!name) {
    alert('Введите название сборки');
    return;
  }
  
  const newInstance = {
    id: instances.length ? Math.max(...instances.map(i => i.id)) + 1 : 1,
    name: name,
    group: group,
    version: version,
    loader: loader,
    ram: ram,
    path: `instances/${name.toLowerCase().replace(/\s/g, '-')}`
  };
  
  instances.push(newInstance);
  renderInstances();
  closeModal('modalCreateInstance');
  
  document.getElementById('instanceName').value = '';
  
  alert(`Сборка "${name}" успешно создана!\n\nПуть: ${newInstance.path}`);
}

function editInstance(id) {
  currentEditingInstanceId = id;
  const inst = instances.find(i => i.id === id);
  if (!inst) return;
  
  document.getElementById('instanceSettingsTitle').textContent = `Настройки: ${inst.name}`;

  document.getElementById('editInstanceGroup').value = inst.group;
  document.getElementById('editInstanceVersion').value = inst.version;
  document.getElementById('editInstanceLoader').value = inst.loader;
  document.getElementById('editInstanceRAM').value = inst.ram;
  document.getElementById('editInstancePath').value = inst.path;
  
  document.getElementById('modalInstanceSettings').classList.add('open');
}

function saveInstanceSettings() {
  if (!currentEditingInstanceId) return;
  
  const inst = instances.find(i => i.id === currentEditingInstanceId);
  if (!inst) return;
  
  inst.name = document.getElementById('editInstanceName').value;
  inst.group = document.getElementById('editInstanceGroup').value;
  inst.version = document.getElementById('editInstanceVersion').value;
  inst.loader = document.getElementById('editInstanceLoader').value;
  inst.ram = parseInt(document.getElementById('editInstanceRAM').value);
  
  renderInstances();
  closeModal('modalInstanceSettings');
  alert('Настройки сохранены!');
}

function deleteCurrentInstance() {
  if (!currentEditingInstanceId) return;
  
  const inst = instances.find(i => i.id === currentEditingInstanceId);
  if (!inst) return;
  
  if (confirm(`Удалить сборку "${inst.name}"?\n\nЭто действие нельзя отменить!`)) {
    instances = instances.filter(i => i.id !== currentEditingInstanceId);
    if (selectedInstanceId === currentEditingInstanceId) {
      selectedInstanceId = null;
    }
    renderInstances();
    closeModal('modalInstanceSettings');
    alert('Сборка удалена');
  }
}

const VERSIONS = {
  releases: [],
  snapshots: [],
  betas: [],
  alphas: [],
  old: []
};

async function loadVersions() {
  try {
    const res = await fetch("https://piston-meta.mojang.com/mc/game/version_manifest_v2.json");
    const data = await res.json();

    VERSIONS.releases = [];
    VERSIONS.snapshots = [];
    VERSIONS.betas = [];
    VERSIONS.alphas = [];
    VERSIONS.old = [];

    data.versions.forEach(v => {
      const date = new Date(v.releaseTime).toLocaleDateString('ru-RU');

      if (v.type === "release") {
        VERSIONS.releases.push([v.id, date]);
      }
      else if (v.type === "snapshot") {
        VERSIONS.snapshots.push([v.id, date]);
      }
      else if (v.type === "old_beta") {
        VERSIONS.betas.push([v.id, date]);
      }
      else if (v.type === "old_alpha") {
        if (v.id.startsWith("c") || v.id.startsWith("rd-") || v.id.startsWith("inf-") || v.id.startsWith("0.31")) {
          VERSIONS.old.push([v.id, date]);
        }
        else if (v.id.startsWith("a")) {
          VERSIONS.alphas.push([v.id, date]);
        }
      }
    });

    VERSIONS.old.unshift(
      ["rd-131655", "13.05.2009"],
      ["rd-132211", "13.05.2009"],
      ["rd-132328", "13.05.2009"],
      ["mc-161607", "16.05.2009"],
      ["mc-161616", "16.05.2009"],
      ["mc-161625", "16.05.2009"],
      ["mc-161648", "16.05.2009"],
      ["c0.0.11a", "17.05.2009"],
      ["c0.0.12a", "17.05.2009"],
      ["c0.0.13a", "21.05.2009"],
      ["c0.0.13a_03", "21.05.2009"],
      ["c0.0.14a_08", "30.05.2009"],
      ["c0.0.15a-1", "31.05.2009"],
      ["c0.0.16a", "11.06.2009"],
      ["c0.0.18a_02", "16.06.2009"],
      ["c0.0.19a_06", "18.06.2009"],
      ["c0.0.20a-c0.27", "23.06.2009"],
      ["c0.0.21a", "24.06.2009"],
      ["c0.0.22a_05", "10.08.2009"],
      ["c0.0.23a_01", "11.08.2009"],
      ["c0.0.24a", "12.08.2009"],
      ["c0.0.25a", "13.08.2009"],
      ["c0.0.28a", "20.08.2009"],
      ["c0.0.29a", "27.08.2009"],
      ["c0.0.30a", "29.08.2009"],
      ["c0.30_01c", "31.08.2009"],
      ["inf-20100618", "18.06.2010"],

function loginUser() {
  const username = prompt('Введите имя пользователя Microsoft:');
  if (username) {
    alert(`Авторизация через Microsoft...\n\nВ реальном лаунчере:\n- Откроется браузер для OAuth\n- Вы войдете через Microsoft\n- Лаунчер получит токен доступа\n\nДобро пожаловать, ${username}!`);
    
    document.querySelectorAll('.un').forEach(el => el.textContent = username);
    document.querySelectorAll('.uh').forEach(el => el.textContent = 'Microsoft аккаунт');
    document.querySelectorAll('.login-btn').forEach(btn => {
      btn.textContent = 'Выйти';
      btn.onclick = () => {
        if (confirm('Выйти из аккаунта?')) {
          document.querySelectorAll('.un').forEach(el => el.textContent = 'Не авторизован');
          document.querySelectorAll('.uh').forEach(el => el.textContent = 'Войдите для игры');
          document.querySelectorAll('.login-btn').forEach(b => {
            b.textContent = 'Войти';
            b.onclick = loginUser;
          });
        }
      };
    });
  }
}

});

  oldest:'Сначала старые'
};

  e.stopPropagation();
